class Student {
  constructor(name, scores) {
    this.name = name;
    this.scores = scores;
  }

  getAverage() {
    let total = 0;

    for (let i = 0; i < this.scores.length; i++) {
      total += this.scores[i];
    }

    return total / this.scores.length;
  }

  getLetterGrade() {
    const average = this.getAverage();

    if (average >= 90) {
      return "A";
    } else if (average >= 80) {
      return "B";
    } else if (average >= 70) {
      return "C";
    } else if (average >= 60) {
      return "D";
    } else {
      return "F";
    }
  }
}

function analyzeClass(students) {
  let classTotal = 0;
  let topStudent = null;
  let highestAverage = -Infinity;

  console.log("------- STUDENT REPORT -------");

  for (const student of students) {
    const average = student.getAverage();
    classTotal += average;

    console.log(
      `${student.name} - Average: ${average.toFixed(1)} | Grade: ${student.getLetterGrade()}`
    );

    if (average > highestAverage) {
      highestAverage = average;
      topStudent = student;
    }
  }

  const classAverage = classTotal / students.length;

  console.log("-----------------------------------");
  console.log(`Class Average: ${classAverage.toFixed(1)}`);
  console.log(`Top Student: ${topStudent.name} with ${highestAverage.toFixed(1)}`);
}

const studentData = [
  { name: "Lany", scores: [85, 90, 92] },
  { name: "Bongbung", scores: [70, 75, 68] },
  { name: "Dutherthe", scores: [95, 98, 92] },
  { name: "Pico", scores: [55, 60, 58] }
];

const students = studentData.map(
  ({ name, scores }) => new Student(name, scores)
);

analyzeClass(students);
