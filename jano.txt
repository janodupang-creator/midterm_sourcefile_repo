class Stack {
  constructor() {
    this.items = [];
  }
  push(element) {
    this.items.push(element);
  }
  pop() {
    return this.items.pop();
  }
  peek() {
    return this.items[this.items.length - 1];
  }
  isEmpty() {
    return this.items.length === 0;
  }
  size() {
    return this.items.length;
  }
  toString() {
    return this.items.join(', ');
  }
}

console.log(`Remaining Actions: ${actionHistory.toString()}`);
console.log(`Stack Size: ${actionHistory.size()}`);
class Queue {
  constructor() {
    this.items = [];
  }
  enqueue(element) {
    this.items.push(element);
  }
  dequeue() {
    return this.items.shift();
  }
  front() {
    return this.items[0];
  }
  isEmpty() {
    return this.items.length === 0;
  }
  size() {
    return this.items.length;
  }
  toString() {
    return this.items.join(', ');
  }
}

class PriorityQueue {
  constructor() {
    this.items = [];
  }
  enqueue(request, priority) {
    const item = { request, priority };
    let insertAt = this.items.length;

    for (let i = 0; i < this.items.length; i += 1) {
      if (priority < this.items[i].priority) {
        insertAt = i;
        break;
      }
    }

    this.items.splice(insertAt, 0, item);
  }
  dequeue() {
    return this.items.shift();
  }
  size() {
    return this.items.length;
  }
  print() {
    if (this.items.length === 0) {
      console.log('(empty)');
      return;
    }

    this.items.forEach((item) => {
      console.log(`${item.request}, Priority ${item.priority}`);
    });
  }
}

class Node {
  constructor(element) {
    this.element = element;
    this.next = null;
  }
}
class LinkedList {
  constructor() {
    this.length = 0;
    this.head = null;
  }
  append(element) {
    const node = new Node(element);
    if (this.head === null) {
      this.head = node;
    } else {
      let current = this.head;
      while (current.next !== null) {
        current = current.next;
      }
      current.next = node;
    }
    this.length += 1;
  }
  insert(position, element) {
    if (!Number.isInteger(position) || position < 0 || position > this.length) {
      return false;
    }

    const node = new Node(element);
    if (position === 0) {
      node.next = this.head;
      this.head = node;
    } else {
      let previous = this.head;
      for (let i = 0; i < position - 1; i += 1) {
        previous = previous.next;
      }
      node.next = previous.next;
      previous.next = node;
    }
    this.length += 1;
    return true;
  }
  removeAt(position) {
    if (!Number.isInteger(position) || position < 0 || position >= this.length) {
      return undefined;
    }

    let removedNode;
    if (position === 0) {
      removedNode = this.head;
      this.head = this.head.next;
    } else {
      let previous = this.head;
      for (let i = 0; i < position - 1; i += 1) {
        previous = previous.next;
      }
      removedNode = previous.next;
      previous.next = removedNode.next;
    }
    this.length -= 1;
    return removedNode.element;
  }
  indexOf(element) {
    let current = this.head;
    let index = 0;
    while (current !== null) {
      if (current.element === element) return index;
      current = current.next;
      index += 1;
    }
    return -1;
  }
  remove(element) {
    const position = this.indexOf(element);
    return position === -1 ? undefined : this.removeAt(position);
  }
  isEmpty() {
    return this.length === 0;
  }
  size() {
    return this.length;
  }
  toString() {
    const elements = [];
    let current = this.head;
    while (current !== null) {
      elements.push(current.element);
      current = current.next;
    }
    return elements.join(', ');
  }
}
